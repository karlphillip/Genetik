# Directory-scoped development settings cannot propagate through genetik_lib or
# affect sibling targets in a parent project. No parent standard/build type override.
if(CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$" AND NOT CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
    if(GENETIK_WARNINGS_AS_ERRORS)
        add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Werror>)
    endif()
    if(GENETIK_THREAD_SANITIZER)
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-fsanitize=thread;-fno-omit-frame-pointer>")
        add_link_options(-fsanitize=thread)
    endif()
    if(GENETIK_SANITIZERS)
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>")
        add_link_options(-fsanitize=address,undefined)
    endif()
elseif(GENETIK_SANITIZERS OR GENETIK_THREAD_SANITIZER)
    message(FATAL_ERROR "Genetik development sanitizers require a GNU-compatible GCC/Clang frontend")
endif()
