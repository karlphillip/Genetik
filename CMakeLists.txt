cmake_minimum_required(VERSION 3.16)
project(GENETIK VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GENETIK_BUILD_EXAMPLES "Build bounded public examples" ON)
option(GENETIK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_subdirectory(genetik)
if(GENETIK_BUILD_EXAMPLES)
    add_executable(genetik_cadence_campaign cadence_campaign.cpp)
    target_link_libraries(genetik_cadence_campaign PRIVATE genetik_lib)
    add_executable(genetik_prime_regression prime_regression.cpp)
    target_link_libraries(genetik_prime_regression PRIVATE genetik_lib)
    add_executable(genetik_regression_search regression_search.cpp)
    target_link_libraries(genetik_regression_search PRIVATE genetik_lib)
    add_executable(genetik_multivariate_demo multivariate_regression.cpp)
    target_link_libraries(genetik_multivariate_demo PRIVATE genetik_lib)
    add_executable(genetik_constant_campaign constant_fitting_campaign.cpp)
    target_link_libraries(genetik_constant_campaign PRIVATE genetik_lib)
    add_executable(genetik_demo demo.cpp)
    target_link_libraries(genetik_demo PRIVATE genetik_lib)
    add_executable(genetik_polynomial_regression polynomial_regression.cpp)
    target_link_libraries(genetik_polynomial_regression PRIVATE genetik_lib)
    add_executable(genetik_expression_demo expression_demo.cpp)
    target_link_libraries(genetik_expression_demo PRIVATE genetik_lib)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
