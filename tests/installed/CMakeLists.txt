cmake_minimum_required(VERSION 3.16)
project(InstalledConsumer LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Deliberately older: the dependency, not this application, must raise the floor.
if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 11)
endif()
if(REQUEST_NO_VERSION)
    set(REQUEST_VERSION "")
elseif(NOT REQUEST_VERSION)
    set(REQUEST_VERSION 0.1)
endif()
if(REQUEST_EXACT)
    set(exact EXACT)
endif()
if(REQUEST_COMPONENT)
    set(components COMPONENTS ${REQUEST_COMPONENT})
endif()
find_package(Genetik ${REQUEST_VERSION} ${exact} CONFIG REQUIRED ${components}
    PATHS "${EXPECTED_PREFIX}" NO_DEFAULT_PATH)
get_filename_component(resolved "${Genetik_DIR}" REALPATH)
get_filename_component(prefix "${EXPECTED_PREFIX}" REALPATH)
string(FIND "${resolved}/" "${prefix}/" offset)
if(NOT offset EQUAL 0)
    message(FATAL_ERROR "resolved a foreign Genetik package: ${resolved}")
endif()
file(WRITE ${CMAKE_BINARY_DIR}/resolved.txt "${resolved}\n${Genetik_VERSION}\n")
get_target_property(public_links Genetik::Genetik INTERFACE_LINK_LIBRARIES)
get_target_property(public_link_options Genetik::Genetik INTERFACE_LINK_OPTIONS)
if(public_links OR public_link_options)
    message(FATAL_ERROR "unexpected dependencies/link flags on this header-only package")
endif()
add_executable(installed_consumer main.cpp other.cpp mixed.c)
target_link_libraries(installed_consumer PRIVATE Genetik::Genetik)
add_subdirectory(quickstarts)
add_subdirectory(oop_consumer)
# These are configuration-logic checks, not qualification of simulated compilers.
function(check_frontend id frontend expected)
    set(CMAKE_CXX_COMPILER_ID ${id})
    set(CMAKE_CXX_COMPILER_FRONTEND_VARIANT ${frontend})
    set(CMAKE_CXX_SIMULATE_ID "")
    set(MSVC FALSE)
    add_library(probe_${id}_${frontend} INTERFACE)
    genetik_require_floating_point(probe_${id}_${frontend} FALSE)
    get_target_property(options probe_${id}_${frontend} INTERFACE_COMPILE_OPTIONS)
    string(FIND "${options}" "${expected}" found)
    if(found EQUAL -1)
        message(FATAL_ERROR "wrong frontend options: ${id}/${frontend}: ${options}")
    endif()
endfunction()
check_frontend(GNU GNU "-fno-fast-math")
check_frontend(Clang GNU "-ffp-contract=off")
check_frontend(Clang MSVC "/fp:strict")
check_frontend(MSVC MSVC "/fp:strict")
