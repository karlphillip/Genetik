cmake_minimum_required(VERSION 3.16)
project(Embedding LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_EXTENSIONS OFF)
# This unrelated parent option must survive inclusion unchanged.
set(BUILD_TESTING ON CACHE BOOL "parent owns testing")
add_subdirectory("${GENETIK_SOURCE}" genetik)
if(NOT CMAKE_CXX_STANDARD EQUAL 14 OR NOT BUILD_TESTING OR CMAKE_CXX_EXTENSIONS)
    message(FATAL_ERROR "Genetik changed parent compiler/testing settings")
endif()
if(TARGET genetik_demo OR TARGET genetik_tests OR TARGET genetik_evaluation_workspace)
    message(FATAL_ERROR "Genetik enabled development extras while embedded")
endif()
add_executable(parent_app main.cpp)
target_link_libraries(parent_app PRIVATE Genetik::Genetik)
add_executable(unrelated unrelated.cpp)
# The canonical alias and legacy source name must denote the same implementation.
get_target_property(aliased Genetik::Genetik ALIASED_TARGET)
if(NOT aliased STREQUAL "genetik_lib")
    message(FATAL_ERROR "legacy source target lost")
endif()
