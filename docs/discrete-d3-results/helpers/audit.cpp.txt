#include "examples/DiscretePrimeResearch.hpp"
#include <iostream>
using namespace discrete_prime;
void predictions(std::ostream& out,const DiscreteExpression& e,std::span<const DiscreteRegressionRow> rows){
 out<<'[';bool first=true;for(const auto& row:rows){if(!first)out<<',';first=false;const auto r=e.evaluate(row.inputs,evaluationPolicy());out<<"{\"n\":"<<row.inputs[0]<<",\"target\":"<<row.target<<",\"status\":"<<static_cast<unsigned>(r.status)<<",\"value\":";if(r.valid())out<<r.value->integer();else out<<"null";out<<",\"visits\":"<<r.nodeVisits<<",\"operations\":"<<r.operations<<",\"iterations\":"<<r.iterations<<'}';}out<<']';
}
int main(int argc,char** argv){try{if(argc!=4)return 2;std::string task=argv[1];unsigned seed=static_cast<unsigned>(std::stoul(argv[2]));auto p=problem(task);auto c=config(task,task=="primality"?20000:5000);RandomContext random(seed,ExecutionConfig{4});auto run=DiscreteEvolutionRun::restore(loadCheckpoint(argv[3]),p,c,random);const auto before=run->checkpoint().payload;const auto work=run->work();const auto rng=random.engine();const auto snapshot=RunCheckpointAccess::population(run->population());std::uint64_t rows=0,visits=0,ops=0,iters=0;std::size_t members=0;
 for(const auto& member:snapshot.members){auto check=p->measure(member.expression());if(check.loss!=member.evaluation())throw std::runtime_error("cached population fitness mismatch");++members;rows+=check.rowsAttempted;visits+=check.nodeVisits;ops+=check.operations;iters+=check.iterations;}
 const auto check=p->measure(snapshot.best.expression());if(check.loss!=snapshot.best.evaluation())throw std::runtime_error("archive fitness mismatch");++members;rows+=check.rowsAttempted;visits+=check.nodeVisits;ops+=check.operations;iters+=check.iterations;
 std::cout<<"{\"task\":"<<quote(task)<<",\"seed\":"<<seed<<",\"cached_entries_verified\":"<<members<<",\"diagnostic_rows\":"<<rows<<",\"diagnostic_visits\":"<<visits<<",\"diagnostic_operations\":"<<ops<<",\"diagnostic_iterations\":"<<iters<<",\"selected\":"<<quote(snapshot.best.text())<<",\"training\":";predictions(std::cout,snapshot.best.expression(),p->rows());std::cout<<",\"held_out\":";predictions(std::cout,snapshot.best.expression(),assessmentRows(task,task=="primality"?64:1001,task=="primality"?255:1200));std::cout<<",\"exterior\":";if(task=="primality")predictions(std::cout,snapshot.best.expression(),assessmentRows(task,256,1024));else std::cout<<"[]";
 if(run->work()!=work||random.engine()!=rng||run->checkpoint().payload!=before)throw std::runtime_error("diagnostic changed run state");std::cout<<",\"training_attempts_added\":0,\"state_unchanged\":true}\n";return 0;
 }catch(const std::exception& e){std::cerr<<e.what()<<'\n';return 1;}}
